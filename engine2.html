// Neuro AI · Google Apps Script
function enviarReporte() {
  const cfg = {
    to: "equipo@tuagencia.com",
    subject: `✦ Reporte IA · ${hoy()}`,
    api: PropertiesService
      .getScriptProperties()
      .getProperty("NEURO_API_KEY")
  };
  const res = UrlFetchApp.fetch(
    "https://api.neuroai.com/v1/report", {
    method: "POST",
    headers: { "Authorization": `Bearer ${cfg.api}` },
    payload: JSON.stringify({ date: hoy() })
  });
  GmailApp.sendEmail(cfg.to, cfg.subject,
    "", { htmlBody: buildHTML(res), name: "Neuro AI" }
  );
}
